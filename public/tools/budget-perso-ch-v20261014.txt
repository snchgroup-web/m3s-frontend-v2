/* Convertit tout le budget personnel : r = valeur d'1 unité de la nouvelle devise dans l'ancienne */
function convertBase(to,r,empty){
  ensureFx();const from=base();if(to===from)return;
  const f=empty?1:1/r, cv=v=>v==null||v===''?v:round2((+v||0)*f);
  if(!empty){
    const nr={};
    ['CHF','EUR','USD'].forEach(c=>{if(c===to)return;if(to==='XOF'&&c==='EUR')return;const o=c===from?1:rateOf(c);if(o)nr[c]={v:+(o*f).toPrecision(6),d:c===from?TODAY:((plan.rates[c]&&plan.rates[c].d)||TODAY)};else nr[c]={v:null,d:''}});
    plan.incomes.forEach(i=>{i.amount=cv(i.amount)});
    plan.cats.forEach(c=>c.lines.forEach(l=>{l.amount=cv(l.amount)}));
    (plan.debts||[]).forEach(d=>{d.total=cv(d.total);d.paidBefore=cv(d.paidBefore)});
    (plan.goals||[]).forEach(g=>{g.target=cv(g.target);g.savedBefore=cv(g.savedBefore)});
    Object.keys(months).forEach(ym=>{const m=months[ym];
      (m.items||[]).forEach(x=>{if(x.orig&&x.orig.cur===to){x.amount=round2(x.orig.amount);delete x.orig}else{const old=+x.amount||0;x.amount=cv(old);if(x.orig)x.orig.rate=+(x.orig.rate*f).toPrecision(8);else x.orig={amount:old,cur:from,rate:+f.toPrecision(8),d:x.date}}});
      if(m.inc)Object.keys(m.inc).forEach(k=>{m.inc[k]=cv(m.inc[k])});
      (m.transfers||[]).forEach(x=>{x.baseSent=cv(x.baseSent);x.baseFees=cv(x.baseFees);if(x.rate)x.rate=+(x.rate*f).toPrecision(8)});
      saveMonth(ym)});
    plan.rates=nr;
  }
  plan.base=to;
  if(!empty)logP('lg_base',{a:curLbl(from),c:curLbl(to),r:String(+(+r).toPrecision(6))});
  savePlan();render();renderChrome();
}

  const bar=(x,v,col,planned)=>{const h=Math.max(0,(v/max)*ih);if(h<.5)return '';const yy=pt+ih-h,r=Math.min(4,h,bw/2);

/* coût réel d'un transfert : reçu comparé au taux officiel, frais compris */
function trCost(x){
  const rc=rateOf(x.cur),rx=rateOf('XOF');let off=null;
  if(x.cur==='EUR')off=XOF_PER_EUR;else if(rc&&rx)off=rc/rx;
  if(!off||!(x.recv>0))return null;
  return 1-x.recv/((x.sent+(x.fees||0))*off);
}
